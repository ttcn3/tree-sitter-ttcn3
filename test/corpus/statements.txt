Real-world: select union with case else clause
===
module HTTP_SelectUnion {
  type union U {
    charstring text,
    octetstring binary
  }

  function f_select(in U p_val) return integer {
    var integer v_Length := 0;

    select union (p_val) {
      case (text)   { v_Length := lengthof(p_val.text); }
      case (binary) { v_Length := lengthof(p_val.binary); }
      case else     { v_Length := -1; }
    }
    return v_Length;
  }
}
