module RealWorldFunctions {
  type component C {
    port integer P;
  }

  function f_behavior(in integer p_count) runs on C {
    var integer i := 0;
    var boolean v_done := false;

    if (p_count == 0) {
      v_done := true;
    }

    for (i := 0;; i := i + 1) {
      P.send(i);
      if (i == 10) {
        break;
      }
    }

    while (v_done == false) {
      i := i + 1;
      if (i == p_count) {
        v_done := true;
      }
    }
  }
}
